
#include <stdio.h>

// declare initial fixed size
#define size 100

// declare the stack and pointer globally
int stack[size];
int top = -1;

void push(int ele) {
    if (top >= size - 1) {
        printf("stack is full\n");
    } else {
        top++;
        stack[top] = ele;
    }
}

void display() {
    if (top == -1) {
        printf("stack is empty\n");
    } else {
        int itr;
        for (itr = top; itr >= 0; itr--) {
            printf("%d\n", stack[itr]);
        }
    }
}

void pop() {
    if (top == -1) {
        printf("the stack is underflow\n");
    } else {
        printf("the element is deleted: %d\n", stack[top]);
        top--;
    }
}

void is_full() {
    if (top >= size - 1)
        printf("the stack is full\n");
    else
        printf("the stack is not full\n");
}

void is_empty() {
    if (top == -1)
        printf("the stack is empty\n");
    else
        printf("the stack is not empty\n");
}

int main() {
    int choice, ele;
    printf("1.push() 2.pop 3.display 4.is_full 5.is_empty -1.exit\n");

    while (1) {
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        if (choice == -1) {
            break;
        }

        switch (choice) {
            case 1:
                printf("Enter element: ");
                scanf("%d", &ele);
                push(ele);
                break;
            case 2:
                pop();
                break;
            case 3:
                display();
                break;
            case 4:
                is_full();
                break;
            case 5:
                is_empty();
                break;
            default:
                printf("Invalid choice\n");
        }
    }
    return 0;
}

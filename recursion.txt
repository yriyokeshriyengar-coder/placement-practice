//recursive function
// this is head recursion

#include<stdio.h>

void natural_no(int num)// function simple sollanumn a process
{
    if (num<=0)
    return 0;
    printf("%d ",num);
    natural_no(num-1);
}


int main()              // only declaraion
{
    int num;
    scanf("%d",&num);
    natural_no(num);
}

}

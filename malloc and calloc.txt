#include <stdio.h>
#include <stdlib.h> 

int main() {
    int num = 7;
    int *ptr = (int *)malloc(sizeof(int) * num);
    int itr;

    if (ptr == NULL) {
        printf("Memory allocation failed!\n");
        return 1; // Exit if memory allocation fails
    }
    printf("Enter %d integers:\n", num);
    for (itr = 0; itr < num; itr++) {
        scanf("%d", ptr + itr); 
    }

    printf("Memory Addresses:\n");
    for (itr = 0; itr < num; itr++) {
        printf("%p ", (void *)&ptr[itr]); 
    }
    printf("\n");

        num = 10;
    int *new_ptr = (int *)realloc(ptr, sizeof(int) * num);

    if (new_ptr == NULL) {
        printf("Memory reallocation failed!\n");
        free(ptr); 
        return 1;
    }
    ptr = new_ptr; // Update ptr to the new memory block

    printf("Enter %d more integers:\n", num - 7); 
    for (itr = 7; itr < num; itr++) { // Start from where we left off (index 7)
        scanf("%d", ptr + itr);
    }

    printf("Data in Reallocated Memory:\n");
    for (itr = 0; itr < num; itr++) {
        printf("%d ", ptr[itr]);
    }
    printf("\n");

    free(ptr); 
    return 0;
}
